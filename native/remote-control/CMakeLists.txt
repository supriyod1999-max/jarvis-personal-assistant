cmake_minimum_required(VERSION 3.25)
project(JarvisRemoteControl LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Microsoft C++ and Windows SDK are required.")
endif()
add_executable(jarvis-remote remote.cpp)
target_compile_features(jarvis-remote PRIVATE cxx_std_20)
target_compile_definitions(jarvis-remote PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(jarvis-remote PRIVATE /W4 /WX /EHsc /sdl /guard:cf)
target_link_options(jarvis-remote PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(jarvis-remote PRIVATE user32)
enable_testing()
add_test(NAME fixed-remote-protocol COMMAND jarvis-remote --self-test)
