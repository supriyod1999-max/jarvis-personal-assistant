cmake_minimum_required(VERSION 3.25)
project(JarvisWindowsSignin LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This component requires Microsoft C++ and the Windows SDK.")
endif()
add_executable(jarvis-signin-build-check build-check.cpp)
target_compile_features(jarvis-signin-build-check PRIVATE cxx_std_20)
target_compile_definitions(jarvis-signin-build-check PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(jarvis-signin-build-check PRIVATE /W4 /WX /EHsc /sdl /guard:cf)
target_link_options(jarvis-signin-build-check PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(jarvis-signin-build-check PRIVATE bcrypt crypt32)
enable_testing()
add_test(NAME windows-native-prerequisites COMMAND jarvis-signin-build-check)
add_executable(jarvis-computer computer.cpp)
target_compile_features(jarvis-computer PRIVATE cxx_std_20)
target_compile_definitions(jarvis-computer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(jarvis-computer PRIVATE /W4 /WX /EHsc /sdl /guard:cf)
target_link_options(jarvis-computer PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(jarvis-computer PRIVATE wtsapi32 advapi32 user32)
add_test(NAME computer-fixed-interface COMMAND jarvis-computer self-test)
add_library(jarvis-vault STATIC vault.cpp)
target_link_libraries(jarvis-vault PUBLIC bcrypt crypt32 shell32 ole32 advapi32 uuid)
add_executable(jarvis-signin-enroll enroll.cpp)
target_link_libraries(jarvis-signin-enroll PRIVATE jarvis-vault credui secur32)
add_executable(jarvis-vault-test vault-test.cpp)
target_link_libraries(jarvis-vault-test PRIVATE jarvis-vault)
add_library(jarvis-credential-provider SHARED provider.cpp approval-notifier.cpp provider.def)
target_link_libraries(jarvis-credential-provider PRIVATE jarvis-vault credui secur32 shlwapi wtsapi32)
add_executable(jarvis-provider-test provider-test.cpp)
target_link_libraries(jarvis-provider-test PRIVATE ole32 uuid)
add_executable(jarvis-proof-test proof-test.cpp)
target_link_libraries(jarvis-proof-test PRIVATE jarvis-vault)
add_executable(jarvis-notifier-test notifier-test.cpp approval-notifier.cpp)
target_link_libraries(jarvis-notifier-test PRIVATE jarvis-vault)
add_executable(jarvis-permissions-test permissions-test.cpp)
target_link_libraries(jarvis-permissions-test PRIVATE jarvis-vault)
foreach(target jarvis-vault jarvis-signin-enroll jarvis-vault-test jarvis-credential-provider jarvis-provider-test jarvis-proof-test jarvis-notifier-test jarvis-permissions-test)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /sdl /guard:cf)
  target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
endforeach()
add_test(NAME synthetic-protected-vault COMMAND jarvis-vault-test)
add_test(NAME provider-rejects-without-approval COMMAND jarvis-provider-test $<TARGET_FILE:jarvis-credential-provider>)
add_test(NAME notification-without-ui-message-pump COMMAND jarvis-notifier-test)
add_test(NAME public-proof-inherits-system-access COMMAND jarvis-permissions-test "${CMAKE_CURRENT_BINARY_DIR}")
